model {  
  # alpha: learning rate (bounded [0,1])
  mu_alpha_raw ~ dnorm(0, 1)
  lambda_alpha ~ dgamma(0.01, 0.01)
  mu_alpha <- phi(mu_alpha_raw)
  
  # theta: inverse temperature/softmax parameter (bounded [0,])
  mu_theta ~ dnorm(0, 0.1)T(0,)
  lambda_theta ~ dgamma(0.01, 0.01)
  
  # Loop through subjects
  for (s in 1:nsubs) {
    
    # Subject-level parameters
    alpha_raw[s] ~ dnorm(mu_alpha_raw, lambda_alpha)
    alpha[s] <- phi(alpha_raw[s])
    
    theta[s] ~ dnorm(mu_theta, lambda_theta)T(0,)
    
    # Initialize expected values for first trial
    Ev[s, 1, 1] <- 0
    Ev[s, 1, 2] <- 0
    Ev[s, 1, 3] <- 0
    Ev[s, 1, 4] <- 0
    
    # Initialize probabilities for first trial
    p[s, 1, 1] <- 0.25
    p[s, 1, 2] <- 0.25
    p[s, 1, 3] <- 0.25
    p[s, 1, 4] <- 0.25
    
    # Loop through trials
    for (t in 2:ntrials[s]) {
      
      # Update expected values for each deck
      for (d in 1:4) {
        
        # Update expected value: only for chosen deck
        Ev_update[s, t, d] <- Ev[s, t-1, d] + alpha[s] * (X[s, t-1] - Ev[s, t-1, d])
        
        # Copy old value if not chosen, update if chosen
        Ev[s, t, d] <- ifelse(d == x[s, t-1], Ev_update[s, t, d], Ev[s, t-1, d])
        
        # Softmax transformation part 1: exponentiate
        exp_p[s, t, d] <- exp(theta[s] * Ev[s, t, d])
      }
      
      # Softmax transformation part 2: normalize
      for (d in 1:4) {
        p[s, t, d] <- exp_p[s, t, d] / sum(exp_p[s, t, ])
      }
      
      # Sample choice from categorical distribution
      x[s, t] ~ dcat(p[s, t, ])
    }
  }
}
