model {
  
  # Hyperpriors for group-level distributions
  # Using probit transformation for parameters bounded [0,1]
  
  # prob_stay_win: probability of staying after a win
  mu_prob_stay_win_raw ~ dnorm(0, 1)
  lambda_prob_stay_win ~ dgamma(1, 0.5) #made slightly more informative (less to explore)
  mu_prob_stay_win <- phi(mu_prob_stay_win_raw)
  
  # prob_stay_lose: probability of staying after a loss
  mu_prob_stay_lose_raw ~ dnorm(0, 1)
  lambda_prob_stay_lose ~ dgamma(1, 0.5) #made slightly more informative (less to explore)
  mu_prob_stay_lose <- phi(mu_prob_stay_lose_raw)
  
  # epsilon: random exploration parameter (trying to account for both low and high exploration)
  mu_epsilon_raw ~ dnorm(0, 1)        
  lambda_epsilon ~ dgamma(2, 1)  
  mu_epsilon <- phi(mu_epsilon_raw)

  
  # Loop through subjects
  for (s in 1:nsubs) {
    
    # Subject-level parameters (probit-transformed)
    prob_stay_win_raw[s] ~ dnorm(mu_prob_stay_win_raw, lambda_prob_stay_win)
    prob_stay_win[s] <- phi(prob_stay_win_raw[s])
    
    prob_stay_lose_raw[s] ~ dnorm(mu_prob_stay_lose_raw, lambda_prob_stay_lose)
    prob_stay_lose[s] <- phi(prob_stay_lose_raw[s])
    
    epsilon_raw[s] ~ dnorm(mu_epsilon_raw, lambda_epsilon)
    epsilon[s] <- phi(epsilon_raw[s])
    
    # Initialize first trial - equal probabilities for all decks
    p[s, 1, 1] <- 0.25
    p[s, 1, 2] <- 0.25
    p[s, 1, 3] <- 0.25
    p[s, 1, 4] <- 0.25
    
    # Initialize first trial for predictions
    p_pred[s, 1, 1] <- 0.25
    p_pred[s, 1, 2] <- 0.25
    p_pred[s, 1, 3] <- 0.25
    p_pred[s, 1, 4] <- 0.25
    
    # Sample first predicted choice
    x_pred[s, 1] ~ dcat(p_pred[s, 1, ])
    
    # Loop through trials starting from trial 2
    for (t in 2:ntrials[s]) {
      
      # Determine if previous outcome was a win or loss
      is_win[s, t] <- ifelse(X[s, t-1] >= 0, 1, 0)
      
      # Determine stay probability based on previous outcome
      stay_prob[s, t] <- ifelse(is_win[s, t] == 1, 
                                 prob_stay_win[s], 
                                 prob_stay_lose[s])
      
      # Calculate choice probabilities for each deck
      for (d in 1:4) {
        
        # If this was the previously chosen deck
        if_stayed[s, t, d] <- ifelse(d == x[s, t-1], 
                                      stay_prob[s, t], 
                                      (1 - stay_prob[s, t]) / 3)
        
        # Mix with random exploration
        p[s, t, d] <- (1 - epsilon[s]) * if_stayed[s, t, d] + epsilon[s] * 0.25
        
        # Same for predicted choices (based on x_pred instead of x)
        if_stayed_pred[s, t, d] <- ifelse(d == x_pred[s, t-1], 
                                           stay_prob[s, t], 
                                           (1 - stay_prob[s, t]) / 3)
        
        p_pred[s, t, d] <- (1 - epsilon[s]) * if_stayed_pred[s, t, d] + epsilon[s] * 0.25
      }
      
      # Sample choice from categorical distribution
      x[s, t] ~ dcat(p[s, t, ])
      
      # Sample predicted choice
      x_pred[s, t] ~ dcat(p_pred[s, t, ])
    }
  }
}
