model {
  
  # Hyperpriors for group-level distributions (separate for each group)
  # Using probit transformation for parameters bounded [0,1]
  # for group param recovery
  
  for (g in 1:n_groups) {
    # prob_stay_win: probability of staying after a win
    mu_prob_stay_win_raw[g] ~ dnorm(0, 1)
    lambda_prob_stay_win[g] ~ dgamma(1, 0.5)
    mu_prob_stay_win[g] <- phi(mu_prob_stay_win_raw[g])
    
    # prob_stay_lose: probability of staying after a loss
    mu_prob_stay_lose_raw[g] ~ dnorm(0, 1)
    lambda_prob_stay_lose[g] ~ dgamma(1, 0.5)
    mu_prob_stay_lose[g] <- phi(mu_prob_stay_lose_raw[g])
    
    # epsilon: random exploration parameter  (trying to account for both low and high exploration)
	mu_epsilon_raw[g] ~ dnorm(-1, 0.5)
	lambda_epsilon[g] ~ dgamma(2, 1)  #group-specific precision!!
	mu_epsilon[g] <- phi(mu_epsilon_raw[g])

  }
  
  # Loop through subjects
  for (s in 1:nsubs) {
    
    # Subject-level parameters drawn from their group distribution
    prob_stay_win_raw[s] ~ dnorm(mu_prob_stay_win_raw[group[s]], lambda_prob_stay_win[group[s]])
    prob_stay_win[s] <- phi(prob_stay_win_raw[s])
    
    prob_stay_lose_raw[s] ~ dnorm(mu_prob_stay_lose_raw[group[s]], lambda_prob_stay_lose[group[s]])
    prob_stay_lose[s] <- phi(prob_stay_lose_raw[s])
    
    epsilon_raw[s] ~ dnorm(mu_epsilon_raw[group[s]], lambda_epsilon[group[s]])
    epsilon[s] <- phi(epsilon_raw[s])
    
    # Initialize first trial - equal probabilities for all decks
    p[s, 1, 1] <- 0.25
    p[s, 1, 2] <- 0.25
    p[s, 1, 3] <- 0.25
    p[s, 1, 4] <- 0.25
    
    # Initialize first trial for predictions
    p_pred[s, 1, 1] <- 0.25
    p_pred[s, 1, 2] <- 0.25
    p_pred[s, 1, 3] <- 0.25
    p_pred[s, 1, 4] <- 0.25
    
    # Sample first predicted choice
    x_pred[s, 1] ~ dcat(p_pred[s, 1, ])
    
    # Loop through trials starting from trial 2
    for (t in 2:ntrials[s]) {
      
      # Determine stay probability based on previous outcome
      # X[s,t-1] is 1 for win, 0 for loss
      stay_prob[s, t] <- X[s, t-1] * prob_stay_win[s] + 
                         (1 - X[s, t-1]) * prob_stay_lose[s]
      
      # Calculate choice probabilities for each deck
      for (d in 1:4) {
        
        # Is this the previously chosen deck?
        is_prev[s, t, d] <- equals(d, x[s, t-1])
        
        # Probability: stay if same deck, switch divided by 3 if different
        if_stayed[s, t, d] <- is_prev[s, t, d] * stay_prob[s, t] + 
                              (1 - is_prev[s, t, d]) * (1 - stay_prob[s, t]) / 3
        
        # Mix with random exploration
        p[s, t, d] <- (1 - epsilon[s]) * if_stayed[s, t, d] + epsilon[s] * 0.25
        
        # Same for predicted choices (based on x_pred instead of x)
        is_prev_pred[s, t, d] <- equals(d, x_pred[s, t-1])
        
        if_stayed_pred[s, t, d] <- is_prev_pred[s, t, d] * stay_prob[s, t] + 
                                   (1 - is_prev_pred[s, t, d]) * (1 - stay_prob[s, t]) / 3
        
        p_pred[s, t, d] <- (1 - epsilon[s]) * if_stayed_pred[s, t, d] + epsilon[s] * 0.25
      }
      
      # Sample choice from categorical distribution
      x[s, t] ~ dcat(p[s, t, ])
      
      # Sample predicted choice
      x_pred[s, t] ~ dcat(p_pred[s, t, ])
    }
  }
}
