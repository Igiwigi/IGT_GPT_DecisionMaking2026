model {

  # -----------------------------
  # Hyperpriors for group-level probabilities
  # -----------------------------
  for (g in 1:n_groups) {
    # Latent probit parameter for win probability
    mu_prob_win_raw[g] ~ dnorm(0, 1)        # group-level mean (probit scale)
    lambda_prob_win[g] ~ dgamma(1, 0.5)     # precision for subjects in group g
    mu_prob_win[g] <- phi(mu_prob_win_raw[g]) # group-level probability of win
  }

  # -----------------------------
  # Subject-level parameters
  # -----------------------------
  for (s in 1:nsubs) {
    # Subject-level latent probit
    win_prob_raw[s] ~ dnorm(mu_prob_win_raw[group[s]], lambda_prob_win[group[s]])
    win_prob[s] <- phi(win_prob_raw[s])      # subject-level probability of win

    # -----------------------------
    # Likelihood for each trial
    # -----------------------------
    for (t in 1:ntrials[s]) {
      X[s, t] ~ dbern(win_prob[s])           # observed outcome: 1 = win, 0 = loss
    }
  }
}
