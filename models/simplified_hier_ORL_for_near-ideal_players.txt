model {

  ####### Group-level (hyper) priors #########

  mu_alpha ~ dnorm(0.75, 0.1)T(0,1)      # mean learning rate
  mu_theta ~ dnorm(0,1)T(0,5)          # mean softmax inverse temperature
  mu_omega_f ~ dnorm(0, 1)              # mean frequency weight
  mu_omega_p ~ dnorm(0, 1)              # mean perseveration weight

  lambda_alpha ~ dgamma(0.01, 0.01)     # precision for subject-level alpha
  lambda_theta ~ dgamma(0.01, 0.01)     # precision for subject-level theta
  lambda_omega_f ~ dgamma(0.01, 0.01)
  lambda_omega_p ~ dgamma(0.01, 0.01)

  ####### Subject-level parameters #########
  for (s in 1:nsubs) {
    alpha[s] ~ dnorm(mu_alpha, lambda_alpha) T(0,1)
    theta[s] ~ dnorm(mu_theta,lambda_theta)T(0,5)
    omega_f[s] ~ dnorm(mu_omega_f, lambda_omega_f)
    omega_p[s] ~ dnorm(mu_omega_p, lambda_omega_p)

    ####### Initialize trial 1 #########
    for (d in 1:4) {
      Ev[s,1,d] <- 0
      Ef[s,1,d] <- 0
      PS[s,1,d] <- 1
      p[s,1,d]  <- 0.25
    }

    ####### Loop over trials #########
    for (t in 2:ntrials[s]) {

      signX[s,t] <- ifelse(X[s,t-1]<0,-1,1)

      for (d in 1:4) {

        # Expected value update (single alpha)
        Ev_update[s,t,d] <- Ev[s,t-1,d] + alpha[s]*(X[s,t-1] - Ev[s,t-1,d])
        Ev[s,t,d] <- ifelse(d==x[s,t-1], Ev_update[s,t,d], Ev[s,t-1,d])

        # Counterfactual / frequency update
        Ef_cho[s,t,d] <- Ef[s,t-1,d] + alpha[s]*(signX[s,t] - Ef[s,t-1,d])
        Ef_not[s,t,d] <- Ef[s,t-1,d] + alpha[s]*(-(signX[s,t]/3) - Ef[s,t-1,d])
        Ef[s,t,d] <- ifelse(d==x[s,t-1], Ef_cho[s,t,d], Ef_not[s,t,d])

        # Perseveration (small decay, optional)
        PS[s,t,d] <- ifelse(d==x[s,t-1], 1/(1+0.05), PS[s,t-1,d]/(1+0.05))

        # Total value
        V[s,t,d] <- Ev[s,t,d] + Ef[s,t,d]*omega_f[s] + PS[s,t,d]*omega_p[s]

        # Softmax numerator
        exp_p[s,t,d] <- exp(theta[s]*V[s,t,d])
      }

      # Softmax denominator
      for (d in 1:4) {
        p[s,t,d] <- exp_p[s,t,d]/sum(exp_p[s,t,])
      }

      # Observed choice
      x[s,t] ~ dcat(p[s,t,])
    }
  }
}
