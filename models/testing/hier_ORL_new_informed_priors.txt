model {
  # ============================================================================
  # GROUP-LEVEL PRIORS (Hyperparameters) - VAGUE BUT SENSIBLE
  # ============================================================================
  
  mu_a_rew ~ dnorm(0, 1)T(0, 1)
  mu_a_pun ~ dnorm(0, 1)T(0, 1)
  mu_K ~ dnorm(0, 1)T(0, )
  mu_theta ~ dunif(0, 5)  # Changed from dnorm to avoid truncation issues
  mu_omega_f ~ dnorm(0, 1)
  mu_omega_p ~ dnorm(0, 1)
  
  # Vague priors on precision (original vague priors)
  lambda_a_rew ~ dgamma(0.01, 0.01)
  lambda_a_pun ~ dgamma(0.01, 0.01)
  lambda_K ~ dgamma(0.01, 0.01)
  lambda_theta ~ dgamma(0.01, 0.01)
  lambda_omega_f ~ dgamma(0.01, 0.01)
  lambda_omega_p ~ dgamma(0.01, 0.01)
  
  # ============================================================================
  # SUBJECT-LEVEL PARAMETERS
  # ============================================================================
  
  for (s in 1:nsubs) {
    
    # Add bounds only where absolutely necessary for model validity
    a_rew[s] ~ dnorm(mu_a_rew, lambda_a_rew)T(0, 1)  # Learning rates must be [0,1]
    a_pun[s] ~ dnorm(mu_a_pun, lambda_a_pun)T(0, 1)
    K[s] ~ dnorm(mu_K, lambda_K)T(0, )  # Decay must be positive
    theta[s] ~ dnorm(mu_theta, lambda_theta)T(0, 10)  # Prevent extreme values
    omega_f[s] ~ dnorm(mu_omega_f, lambda_omega_f)
    omega_p[s] ~ dnorm(mu_omega_p, lambda_omega_p)
    
    # ========================================================================
    # INITIALIZE TRIAL 1 VALUES
    # ========================================================================
    
    Ev[s, 1, 1] <- 0
    Ev[s, 1, 2] <- 0
    Ev[s, 1, 3] <- 0
    Ev[s, 1, 4] <- 0
    
    Ef[s, 1, 1] <- 0
    Ef[s, 1, 2] <- 0
    Ef[s, 1, 3] <- 0
    Ef[s, 1, 4] <- 0
    
    PS[s, 1, 1] <- 1
    PS[s, 1, 2] <- 1
    PS[s, 1, 3] <- 1
    PS[s, 1, 4] <- 1
    
    p[s, 1, 1] <- 0.25
    p[s, 1, 2] <- 0.25
    p[s, 1, 3] <- 0.25
    p[s, 1, 4] <- 0.25
    
    # ========================================================================
    # TRIAL-BY-TRIAL UPDATE LOOP
    # ========================================================================
    
    for (t in 2:ntrials[s]) {
      
      signX[s, t] <- ifelse(X[s, t-1] < 0, -1, 1)
      
      for (d in 1:4) {
        
        # -------- Updating expected values ------------------------
        Ev_update[s, t, d] <- ifelse(X[s, t-1] >= 0,
          Ev[s, t-1, d] + a_rew[s] * (X[s, t-1] - Ev[s, t-1, d]), 
          Ev[s, t-1, d] + a_pun[s] * (X[s, t-1] - Ev[s, t-1, d])
        )
        
        Ev[s, t, d] <- ifelse(d == x[s, t-1], Ev_update[s, t, d], Ev[s, t-1, d])
        
        # -------- Updating expected frequencies ------------------------
        Ef_cho[s, t, d] <- ifelse(X[s, t-1] >= 0, 
          Ef[s, t-1, d] + a_rew[s] * (signX[s, t] - Ef[s, t-1, d]),
          Ef[s, t-1, d] + a_pun[s] * (signX[s, t] - Ef[s, t-1, d])
        )
        
        Ef_not[s, t, d] <- ifelse(X[s, t-1] >= 0, 
          Ef[s, t-1, d] + a_pun[s] * (-(signX[s, t]/3) - Ef[s, t-1, d]),
          Ef[s, t-1, d] + a_rew[s] * (-(signX[s, t]/3) - Ef[s, t-1, d])
        )
        
        Ef[s, t, d] <- ifelse(d == x[s, t-1], Ef_cho[s, t, d], Ef_not[s, t, d])
        
        # -------- Updating perseveration ------------------------
        PS[s, t, d] <- ifelse(x[s, t-1] == d, 1/(1 + K[s]), PS[s, t-1, d]/(1 + K[s]))
        
        # Calculate total value
        V[s, t, d] <- Ev[s, t, d] + Ef[s, t, d] * omega_f[s] + PS[s, t, d] * omega_p[s]
        
      }
      
      # ========================================================================
      # CRITICAL FIX: NUMERICALLY STABLE SOFTMAX
      # ========================================================================
      
      # Subtract maximum value before exponentiation to prevent overflow
      max_V[s, t] <- max(V[s, t, ])
      
      for (d in 1:4) {
        exp_p[s, t, d] <- exp(theta[s] * (V[s, t, d] - max_V[s, t]))
      }
      
      for (d in 1:4) {
        p[s, t, d] <- exp_p[s, t, d] / sum(exp_p[s, t, ])
      }
      
      x[s, t] ~ dcat(p[s, t, ])
      
    }
  }
}
