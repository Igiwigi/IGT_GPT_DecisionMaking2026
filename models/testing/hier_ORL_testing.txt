model {

  ##########################################################
  # GROUP-LEVEL PRIORS
  ##########################################################

  # Learning rates (bounded 0-1)
  mu_a_rew ~ dnorm(0.5, 2) T(0,1)
  mu_a_pun ~ dnorm(0.5, 2) T(0,1)

  # Decay (K > 0)
  mu_K ~ dnorm(0.5, 2) T(0,)

  # Temperature (theta, constrained 0-5)
  mu_theta ~ dnorm(2, 1) T(0,5)

  # Frequency and perseveration weights
  mu_omega_f ~ dnorm(0, 1)
  mu_omega_p ~ dnorm(0, 1)

  # ========================================================
  # HYPERPRIORS FOR SUBJECT-LEVEL VARIABILITY
  # ========================================================

  lambda_a_rew   ~ dgamma(1, 1)
  lambda_a_pun   ~ dgamma(1, 1)
  lambda_K       ~ dgamma(1, 1)
  lambda_theta   ~ dgamma(1, 1)
  lambda_omega_f ~ dgamma(1, 1)
  lambda_omega_p ~ dgamma(1, 1)

  ##########################################################
  # SUBJECT-LEVEL PARAMETERS
  ##########################################################

  for (s in 1:nsubs) {

    # Non-centered parameters for theta
    theta_raw[s] ~ dnorm(0,1)

    # Transformations for all subject-level parameters
    a_rew[s] <- ilogit(mu_a_rew + theta_raw[s]/sqrt(lambda_a_rew))   # bounded 0-1
    a_pun[s] <- ilogit(mu_a_pun + theta_raw[s]/sqrt(lambda_a_pun))   # bounded 0-1
    K[s]     ~ dnorm(mu_K, lambda_K) T(0,)                            # positive
    theta_unbounded[s] <- mu_theta + theta_raw[s]/sqrt(lambda_theta)
    theta[s] <- max(0.0001, min(theta_unbounded[s], 5))               # bounded 0-5
    omega_f[s] ~ dnorm(mu_omega_f, lambda_omega_f)
    omega_p[s] ~ dnorm(mu_omega_p, lambda_omega_p)

    ##########################################################
    # INITIALIZATION FOR TRIAL 1
    ##########################################################
    for (d in 1:4) {
      Ev[s,1,d] <- 0
      Ef[s,1,d] <- 0
      PS[s,1,d] <- 1
      p[s,1,d]  <- 0.25
    }

    ##########################################################
    # TRIAL LOOP
    ##########################################################
    for (t in 2:ntrials[s]) {

      # Sign of previous outcome
      signX[s,t] <- ifelse(X[s,t-1] < 0, -1, 1)

      for (d in 1:4) {

        # -----------------------
        # Expected value update
        # -----------------------
        Ev_update[s,t,d] <- ifelse(
          X[s,t-1] >= 0,
          Ev[s,t-1,d] + a_rew[s] * (X[s,t-1] - Ev[s,t-1,d]),
          Ev[s,t-1,d] + a_pun[s] * (X[s,t-1] - Ev[s,t-1,d])
        )

        Ev[s,t,d] <- ifelse(d == x[s,t-1],
                             Ev_update[s,t,d],
                             Ev[s,t-1,d])

        # -----------------------
        # Frequency update
        # -----------------------
        Ef_cho[s,t,d] <- ifelse(
          X[s,t-1] >= 0,
          Ef[s,t-1,d] + a_rew[s] * (signX[s,t] - Ef[s,t-1,d]),
          Ef[s,t-1,d] + a_pun[s] * (signX[s,t] - Ef[s,t-1,d])
        )

        Ef_not[s,t,d] <- ifelse(
          X[s,t-1] >= 0,
          Ef[s,t-1,d] + a_pun[s] * (-(signX[s,t]/3) - Ef[s,t-1,d]),
          Ef[s,t-1,d] + a_rew[s] * (-(signX[s,t]/3) - Ef[s,t-1,d])
        )

        Ef[s,t,d] <- ifelse(d == x[s,t-1],
                             Ef_cho[s,t,d],
                             Ef_not[s,t,d])

        # -----------------------
        # Perseveration
        # -----------------------
        PS[s,t,d] <- ifelse(
          x[s,t-1] == d,
          1 / (1 + K[s]),
          PS[s,t-1,d] / (1 + K[s])
        )

        # -----------------------
        # Total value
        # -----------------------
        V[s,t,d] <- Ev[s,t,d] +
                    Ef[s,t,d] * omega_f[s] +
                    PS[s,t,d] * omega_p[s]
      }

      # -----------------------
      # Softmax (numerically stable)
      # -----------------------
      max_V[s,t] <- max(V[s,t,])
      for (d in 1:4) {
        exp_p[s,t,d] <- exp(theta[s] * (V[s,t,d] - max_V[s,t]))
      }
      for (d in 1:4) {
        p[s,t,d] <- exp_p[s,t,d] / sum(exp_p[s,t,])
      }

      # -----------------------
      # Likelihood
      # -----------------------
      x[s,t] ~ dcat(p[s,t,])
    } # end trial loop
  } # end subject loop
} # end model
